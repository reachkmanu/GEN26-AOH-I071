% SLIDE 1: PROBLEM STATEMENT
% ============================================================
\section{Problem Statement}
\begin{frame}{Problem Statement}
  \GenesisCornerLogo
  \begin{beamercolorbox}[rounded=true,sep=0.35cm,wd=\dimexpr\textwidth-0.7cm\relax]{block body}
    \textbf{Thermally-Managed Solid-State Battery Module with Embedded Micro-Sensors.}
  \end{beamercolorbox}
  \vspace{0.25cm}
  \begin{itemize}
    \item During DC fast-charging, cells heat unevenly and hot spots form at cell interfaces and tabs, where pack-level cooling cannot reach in time.
    \item Many packs rely on sparse, module-level temperature sensing, so local cell-to-cell differences can go unnoticed until they grow large \src{Liu et al., 2019}.
    \item The aim is a battery module that links per-cell micro-sensing to targeted cooling. The first demonstrator uses conventional lithium-ion cells to validate the thermal-control architecture; solid-state compatibility is a future research objective, not a demonstrated capability.
  \end{itemize}
\end{frame}

% ============================================================
% SLIDE 2: PROBLEM UNDERSTANDING
% ============================================================
\section{Problem Understanding}
\begin{frame}{Problem Understanding}
  \GenesisCornerLogo
  \begin{itemize}
    \item \textbf{Who is affected:} EV owners and fleet operators (cabs, delivery vehicles, buses) that fast-charge frequently, and the manufacturers who design and validate battery thermal systems.
    \item \textbf{Root causes:} uneven current distribution, differences in cell internal resistance, and limited heat paths between cells; resistive heating scales with current squared ($I^2R$), so faster charging sharply raises heat.
    \item \textbf{Consequences:} cells age unevenly and the weakest cell limits pack life; the BMS throttles charging power as a precaution, and the risk of thermal runaway rises \src{Zhang et al., 2022}.
    \item \textbf{Why now:} high-power DC charging (of the order of 150~kW) is becoming common, and a commonly cited operating window for conventional lithium-ion is roughly 25--40~\textdegree C, with cell-to-cell spread kept within a few degrees as an engineering target; the exact limits depend on cell chemistry, construction and charge rate \src{Akbarzadeh et al., 2021}.
  \end{itemize}
\end{frame}

% ============================================================
% SLIDE 3: EXISTING GAPS & CHALLENGES
% ============================================================
\section{Existing Gaps \& Challenges}
\begin{frame}{Existing Gaps \& Challenges}
  \GenesisCornerLogo
  \begin{itemize}
    \item \textbf{Sparse sensing:} where sensing is placed only at selected module locations, hot spots at cell interfaces and tabs can be detected late; cell-level monitoring remains an active research area \src{Liu et al., 2019}.
    \item \textbf{Uniform cooling:} cold plates and cooling ribbons tend to cool the whole pack with little local control, and air cooling is limited at fast-charge heat loads \src{Zhang et al., 2022; Xia et al., 2017}.
    \item \textbf{Passive buffering limits:} phase change materials help with uniformity but suffer from low thermal conductivity, latent-heat saturation and leakage \src{Goli et al., 2014}.
    \item \textbf{Largely reactive control:} many conventional systems raise cooling toward peak power once thresholds are approached; the proposed design should be compared against a clearly defined baseline controller.
    \item \textbf{Opportunity:} low-cost systems that tightly couple per-cell sensing to targeted cooling in real time are not yet common in production (subject to market and prior-art review).
  \end{itemize}
\end{frame}

% ============================================================
% SLIDE 4: OUR PROPOSED SOLUTION
% ============================================================
\section{Our Proposed Solution}
\begin{frame}{Our Proposed Solution}
  \GenesisCornerLogo
  \begin{itemize}
    \item \textbf{Core idea:} a battery-module enclosure that combines a mini-channel liquid cold plate, intended to be divided into independently addressable cooling zones, with a per-cell temperature micro-sensor array.
    \item \textbf{How it works:} the micro-sensors map cell temperatures in near real time; a controller is intended to route more coolant to the hottest zones while keeping the rest at a low-power baseline.
    \item \textbf{Hybrid buffering:} a graphene-enhanced phase change material layer is proposed between the cells and the cold plate to absorb short, high-rate thermal peaks \src{Goli et al., 2014; Hussain et al., 2018}.
  \end{itemize}
  \vspace{0.1cm}
  \begin{beamercolorbox}[rounded=true,sep=0.2cm,wd=\dimexpr\textwidth-0.5cm\relax]{block body}
    \small\textbf{Central testable proposition:} can closed-loop, zone-specific coolant control, informed by distributed sensing and supported by a PCM buffer, reduce peak temperature and total cooling energy versus conventional uniform cooling under the same heat load?
  \end{beamercolorbox}
\end{frame}

% SLIDE 6: INNOVATION & UNIQUENESS
% ============================================================
\section{Innovation \& Uniqueness}
\begin{frame}{Innovation \& Uniqueness}
  \GenesisCornerLogo
  \parbox{\textwidth}{\scriptsize Figures below are published results under each study's specific conditions; whether they transfer to the proposed module is still to be tested.}
  \vspace{0.12cm}
  \begin{itemize}
    \item \textbf{Per-cell, in-situ sensing} instead of sparse module-level readings, to detect hot spots earlier \src{Liu et al., 2019}.
    \item \textbf{Zonal, micro-valved cooling} aims to direct coolant where heat is; a hybrid liquid-plus-PCM cold plate cut pumping energy by up to about 30\% versus a conventional aluminium plate in its own design and test conditions \src{Akbarzadeh et al., 2021}.
    \item \textbf{Graphene-enhanced PCM buffer:} graphene raised PCM thermal conductivity by more than two orders of magnitude as a material property \src{Goli et al., 2014}; graphene-coated nickel foam with paraffin lowered cell surface-temperature rise by about 17\% relative to nickel foam at a 1.7~A discharge \src{Hussain et al., 2018}.
    \item \textbf{Intended differentiator (to be validated):} one low-cost module that couples per-cell sensing to targeted cooling; novelty is to be established against defined baselines and a prior-art review.
  \end{itemize}
\end{frame}

% ============================================================
% SLIDE 7: FEASIBILITY & SCALABILITY
% ============================================================
\section{Feasibility \& Scalability}
\begin{frame}{Feasibility \& Scalability}
  \GenesisCornerLogo
  \begin{itemize}
    \item \textbf{Feasibility:} the demonstrator can be built from off-the-shelf parts with lab access and a mentor; prior work reports distributed forced convection as a cost-effective route to uniform temperature \src{Karimi and Li, 2013} and hybrid liquid-plus-PCM plates tested experimentally \src{Akbarzadeh et al., 2021}.
    \item \textbf{Validation path:} first in thermal--CFD simulation, then on a small 4--6 cell module, compared against conventional uniform-flow cooling at the same heat input, measuring cooling energy and not only temperature.
    \item \textbf{Success criteria (provisional targets):} tighter cell-to-cell spread and lower peak temperature at equal or lower cooling energy than the baseline.
    \item \textbf{Scalability:} the zoned module is intended to be modular; a small demonstrator does not by itself establish vehicle-pack scalability, durability, serviceability or safety, which require further work.
    \item \textbf{Risks and mitigations:} per-cell wiring cost and failure points (multiplexed or flexible printed-circuit arrays); PCM leakage (shape-stabilised or composite PCM); coolant sealing and service access.
  \end{itemize}
\end{frame}

\section{Use Cases}
\begin{frame}{Use Cases {\normalsize\mdseries (compulsorily 3 use cases must be there)}}
  \GenesisCornerLogo
  \begin{itemize}
    \item \textbf{Use case 1 --- Fast-charging fleet vehicles:} taxis, delivery vans and buses that fast-charge several times a day; targeted cooling is intended to hold the cell-to-cell spread tight across repeated DC fast charges, with the aim of extending pack life and reducing downtime (potential outcomes, to be validated).
    \item \textbf{Use case 2 --- Manufacturer validation rig:} battery-pack makers could use the per-cell sensing and zonal cooling module as a test platform to study thermal behaviour of new cell formats, including pilot solid-state cells.
    \item \textbf{Use case 3 --- High-ambient, high-C-rate operation:} vehicles in hot climates or at high discharge rates; proactive zonal cooling aims to limit local hot spots and reduce precautionary throttling of charge or discharge power.
  \end{itemize}
\end{frame}

\section{Impacts and Benefits}
\begin{frame}{Impacts and Benefits}
  \GenesisCornerLogo
  \parbox{\textwidth}{\scriptsize The items below are expected benefits contingent on validation, not outcomes demonstrated by the prototype.}
  \vspace{0.12cm}
  \begin{itemize}
    \item \textbf{For users:} aims to enable faster charging and longer battery life, which would lower the cost per kilometre for drivers and fleets.
    \item \textbf{For manufacturers:} potential for fewer warranty claims from uneven aging, and a thermal-control architecture that could inform future solid-state packs.
    \item \textbf{Safety framing:} the system aims to improve thermal monitoring and temperature control; lowering peak temperature and improving uniformity supports safety but does not, by itself, demonstrate prevention of thermal runaway.
    \item \textbf{Projected targets (to be validated):} cell-to-cell spread within about 5~\textdegree C and peak temperature below about 40~\textdegree C during fast-charge-like loads, at equal or lower cooling energy than uniform cooling \src{cf. Akbarzadeh et al., 2021}.
  \end{itemize}
\end{frame}
